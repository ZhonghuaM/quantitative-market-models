cmake_minimum_required(VERSION 3.16)
project(portfolio_risk_agent VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(QFA_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(QFA_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  else()
    message(FATAL_ERROR "QFA_SANITIZERS requires Clang or GCC")
  endif()
endif()

add_library(qfa_numerics STATIC native/src/core.cpp native/src/rng.c)
target_include_directories(qfa_numerics PUBLIC native/include)
add_executable(qfa_engine native/src/main.cpp)
target_link_libraries(qfa_engine PRIVATE qfa_numerics)
if(MSVC)
  target_compile_options(qfa_numerics PRIVATE /W4)
  target_compile_options(qfa_engine PRIVATE /W4)
else()
  target_compile_options(qfa_numerics PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(qfa_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(qfa_native_tests native/tests/test_core.cpp)
  target_link_libraries(qfa_native_tests PRIVATE qfa_numerics)
  add_test(NAME native_numerics COMMAND qfa_native_tests)
  add_test(NAME cli_unknown_command COMMAND qfa_engine unknown)
  set_tests_properties(cli_unknown_command PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_unknown_flag COMMAND qfa_engine risk --unknown 1)
  set_tests_properties(cli_unknown_flag PROPERTIES WILL_FAIL TRUE)
endif()
